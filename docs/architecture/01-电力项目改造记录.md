#### 2. 双连接池架构与 SQLAlchemy 2.0 异步配置

在 `backend/src/core/database.py` 中实现**业务 ORM 与 LangGraph Checkpointer 的物理隔离**：

Python

```
from sqlalchemy.ext.asyncio import create_async_engine, async_sessionmaker, AsyncSession
from psycopg_pool import AsyncConnectionPool
from contextlib import asynccontextmanager

PG_URI = "postgresql+asyncpg://ai_admin:${DB_PASSWORD}@localhost:5432/power_dispatch_ai"
PSYCOPG_URI = "postgresql://ai_admin:${DB_PASSWORD}@localhost:5432/power_dispatch_ai"

# 1. 业务 ORM 连接池：专用于会话映射、向量检索、审计 Trace
async_engine = create_async_engine(
    PG_URI,
    pool_size=20,
    max_overflow=10,
    pool_recycle=1800,
    pool_pre_ping=True,
    echo=False
)
AsyncSessionLocal = async_sessionmaker(async_engine, expire_on_commit=False, class_=AsyncSession)

# 2. LangGraph 状态机独立连接池：专用于 AsyncPostgresSaver，防写锁耗尽业务连接
checkpointer_pool = AsyncConnectionPool(
    conninfo=PSYCOPG_URI,
    max_size=20,
    kwargs={"autocommit": True}
)

async def get_db_session() -> AsyncSession:
    async with AsyncSessionLocal() as session:
        yield session
```

### 第二阶段：AI 专属库表结构设计（仅存 AI 资产）

#### 1. 核心表结构定义（`backend/src/models/ai_schema.py`）

坚决不存“工单主数据”，仅存 **Thread 映射**、**消息快照**、**向量条文** 与 **审计 Trace**：

Python

```
import uuid
from datetime import datetime
from sqlalchemy import String, Text, DateTime, ForeignKey, Index
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column, relationship
from pgvector.sqlalchemy import Vector

class Base(DeclarativeBase):
    pass

class AIThreadMapping(Base):
    """会话映射表：将前端会话与外部业务工单挂接"""
    __tablename__ = "ai_thread_mappings"

    thread_id: Mapped[str] = mapped_column(String(64), primary_key=True)  # UUIDv7 / 业务单派生 ID
    session_id: Mapped[str] = mapped_column(String(64), index=True)       # 前端 UI 对话框 ID (支持单会话多工单)
    creator_id: Mapped[str] = mapped_column(String(64), index=True)       # 发起人用户 ID
    creator_role: Mapped[str] = mapped_column(String(32))                 # APPLICANT / DISPATCHER
    assigned_role: Mapped[str] = mapped_column(String(32), default="APPLICANT") # 当前协同处理角色
    status: Mapped[str] = mapped_column(String(32), default="DRAFT")      # DRAFT | WAITING | PERMITTED | REJECTED
    biz_order_id: Mapped[str | None] = mapped_column(String(64), index=True) # 外部 OMS 工单句柄
    created_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow)
    updated_at: Mapped[datetime] = mapped_column(DateTime, default=datetime.utcnow, onupdate=datetime.utcnow)

class AIChatMessage(Base):
    """消息与 A2UI 卡片快照表：记录完整交互与重试载荷"""
    __tablename__ = "ai_chat_messages"

    id: Mapped[str] = mapped_column(String(64), primary_key=True)
    thread_id: Mapped[str] = mapped_column(String(64), ForeignKey("ai_thread_mappings.thread_id", ondelete="CASCADE"))
    role: Mapped[str] = mapped_column(String(16))      # user / agent / system
    kind: Mapped[str] = mapped_column(String(16))      # text / card
    content: Mapped[str | None] = mapped_column(Text)
    card_type: Mapped[str | None] = mapped_column(String(32))
    card_payload: Mapped[str | None] = mapped_column(Text)  # JSON 字符串
    retry_payload: Mapped[str | None] = mapped_column(Text) # 请求快照，供自包含精准重试

class AISafetyRegulation(Base):
    """安规向量知识库表"""
    __tablename__ = "ai_safety_regulations"

    id: Mapped[uuid.UUID] = mapped_column(primary_key=True, default=uuid.uuid4)
    regulation_code: Mapped[str] = mapped_column(String(64), index=True) # 如 "DL-408-2021"
    voltage_level: Mapped[str] = mapped_column(String(32), index=True)   # "110kV" / "220kV" / "500kV"
    content: Mapped[Text] = mapped_column(Text)
    embedding: Mapped[Vector] = mapped_column(Vector(1536))              # 向量维度

# 构建 HNSW 余弦相似度索引
Index(
    "idx_regulation_embedding_hnsw",
    AISafetyRegulation.embedding,
    postgresql_using="hnsw",
    postgresql_with={"m": 16, "ef_construction": 64},
    postgresql_ops={"embedding": "vector_cosine_ops"}
)
```

```
import jwt
from datetime import datetime, timedelta

JWT_SECRET = "MOCK_ENTERPRISE_KEY_2026"
ALGORITHM = "HS256"

# 预设正式业务账号
MOCK_USERS = {
    "applicant_01": {"name": "张三(运检二班)", "role": "APPLICANT", "dept": "腊山运检工区"},
    "dispatcher_lead": {"name": "李调度长", "role": "DISPATCHER", "dept": "地调控制中心"}
}

def create_access_token(user_id: str) -> str:
    user = MOCK_USERS.get(user_id)
    if not user:
        raise ValueError("用户不存在")
    payload = {
        "sub": user_id,
        "name": user["name"],
        "role": user["role"],
        "dept": user["dept"],
        "exp": datetime.utcnow() + timedelta(hours=12)
    }
    return jwt.encode(payload, JWT_SECRET, algorithm=ALGORITHM)

def verify_token(token: str) -> dict:
    return jwt.decode(token, JWT_SECRET, algorithms=[ALGORITHM])
```

#### 3. 调度自动化防腐适配器（`scada_adapter.py`）

负责生成合规且带数字签名 Hash 的调度令凭证（代替前端纯 CSS 印章）：

Python

```
import hashlib
from datetime import datetime

class SCADAGatewayAdapter:
    @staticmethod
    async def issue_dispatch_permit(biz_order_id: str, reviewer_id: str) -> dict:
        """签发具备法律效力的调度令并生成 CA 防伪签名摘要"""
        permit_no = f"DISP-{datetime.now().strftime('%Y%m%d')}-{biz_order_id[-4:]}"
        raw_sign = f"{permit_no}|{biz_order_id}|{reviewer_id}|APPROVED"
        ca_signature_hash = hashlib.sha256(raw_sign.encode()).hexdigest()
        
        return {
            "permit_no": permit_no,
            "status": "PERMITTED",
            "ca_signature_hash": ca_signature_hash,
            "issued_at": datetime.utcnow().isoformat()
        }
```

### 第四阶段：FastAPI 业务层与 LangGraph 改造

#### 1. Checkpointer 生产级挂载（`main.py`）

Python

```
from fastapi import FastAPI
from contextlib import asynccontextmanager
from langgraph.checkpoint.postgres.aio import AsyncPostgresSaver
from core.database import checkpointer_pool

checkpointer = None

@asynccontextmanager
async def lifespan(app: FastAPI):
    global checkpointer
    async with checkpointer_pool:
        # 使用独立的 psycopg 连接池挂载检查点
        checkpointer = AsyncPostgresSaver(checkpointer_pool)
        await checkpointer.setup()  # 自动建表 checkpoints / checkpoint_writes
        yield

app = FastAPI(lifespan=lifespan)
```

#### 2. 多工单派生与状态机恢复（纯参数注入方案 A）

- **会话物理隔离**：根据当前登录人角色，在 `ai_thread_mappings` 表精准隔离列表。

- **单会话多次申办**：检测到已有激活工单时，**自动派生新 `thread_id`**，确保一个状态机只对应一笔调度令。

- **状态机唤醒**：彻底移除 `as_node`，调度长批准时仅传入参数：

  Python

### 第五阶段：前端高性能渲染与防伪落地
